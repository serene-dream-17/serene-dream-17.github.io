const nav = document.getElementById('nav');
const search = document.getElementById('search');
const empty = document.getElementById('empty');
const norm = s => s.toLocaleLowerCase().trim();
function iconSvg(name){
return `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.7" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true">${icons[name]||icons.layers}</svg>`;
}
function render(){
const q=norm(search.value);
tree.innerHTML='';nav.innerHTML='';
let visibleDomains=0,groupsCount=0,itemsCount=0;
data.forEach((section,idx)=>{
const groups=section.groups.map(g=>({name:g[0],items:g[1].split('、').filter(Boolean)})).map(g=>({
...g,items:g.items.filter(item=>!q||norm(item).includes(q)||norm(g.name).includes(q)||norm(section.title).includes(q))
})).filter(g=>!q||g.items.length);
const titleMatch=q&&norm(section.title).includes(q);
const filteredGroups=q&&!titleMatch?groups:section.groups.map(g=>({name:g[0],items:g[1].split('、').filter(Boolean)}));
const finalGroups=q?(titleMatch?section.groups.map(g=>({name:g[0],items:g[1].split('、').filter(Boolean)})):groups):section.groups.map(g=>({name:g[0],items:g[1].split('、').filter(Boolean)}));
const visible= !q || titleMatch || finalGroups.length>0;
if(!visible)return;
visibleDomains++;groupsCount+=finalGroups.length;itemsCount+=finalGroups.reduce((n,g)=>n+g.items.length,0);
const navButton=document.createElement('button');
navButton.className='nav-item';navButton.innerHTML=`<span class="nav-num">${String(idx+1).padStart(2,'0')}</span><span>${section.title.replace(/^\\d+\\. /,'')}</span>`;
navButton.addEventListener('click',()=>{
const target=document.getElementById('section-'+idx);
if(target){target.classList.remove('collapsed');target.scrollIntoView({behavior:'smooth',block:'start'});}
document.querySelectorAll('.nav-item').forEach(n=>n.classList.remove('active'));navButton.classList.add('active');
});
nav.appendChild(navButton);
const el=document.createElement('section');
el.className='section';el.id='section-'+idx;
if(!q && idx>2)el.classList.add('collapsed');
el.style.setProperty('--tone',section.tone);el.style.setProperty('--tint',section.tint);
const body=finalGroups.map(g=>{
const tags=g.items.map(item=>{
const isMatch=q&&norm(item).includes(q);
return `<span class="tag ${isMatch?'match':''}" title="${item.replace(/"/g,'&quot;')}">${item}</span>`;
}).join('');
return `<article class="group"><h3>${g.name}</h3><div class="tags">${tags}</div></article>`;
}).join('');
el.innerHTML=`<div class="section-head" role="button" tabindex="0" aria-expanded="${!el.classList.contains('collapsed')}">
<div class="section-icon">${iconSvg(section.icon)}</div>
<div class="section-heading"><h2>${section.title}</h2><p>${section.desc}</p></div>
<span class="section-count">${finalGroups.length} 个主题</span>
<span class="chevron">${iconSvg('layers').replace('<svg','<svg style="width:17px;height:17px"')}</span>
</div><div class="section-body"><div class="group-grid">${body}</div></div>`;
const head=el.querySelector('.section-head');
const toggle=()=>{el.classList.toggle('collapsed');head.setAttribute('aria-expanded',String(!el.classList.contains('collapsed')))};
head.addEventListener('click',toggle);head.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){e.preventDefault();toggle();}});
tree.appendChild(el);
});
document.getElementById('domainCount').textContent=visibleDomains;
document.getElementById('groupCount').textContent=groupsCount;
document.getElementById('techCount').textContent=itemsCount;
empty.style.display=visibleDomains?'none':'block';
}

document.getElementById('collapseAll').addEventListener('click',()=>document.querySelectorAll('.section').forEach(s=>s.classList.add('collapsed')));
document.getElementById('clearSearch').addEventListener('click',()=>{search.value='';render();search.focus();});
render();
